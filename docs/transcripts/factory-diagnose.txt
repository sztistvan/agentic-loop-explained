# Basic cleaning: convert timestamp
clean = df.copy()
clean['timestamp'] = pd.to_datetime(clean['timestamp'], errors='coerce')
# remove exact duplicates
clean = clean.drop_duplicates()
# standardize status whitespace/lower
clean['status'] = clean['status'].astype(str).str.strip().str.lower()
# compute attention score based on fault/maintenance rate and severity proxy (vibration/temperature)
summary = clean.groupby('machine_id').agg(
    n_obs=('status','size'),
    fault_events=('status', lambda s: (s=='fault').sum()),
    maintenance_events=('status', lambda s: (s=='maintenance').sum()),
    running_events=('status', lambda s: (s=='running').sum()),
    fault_rate=('status', lambda s: (s=='fault').mean()),
    maintenance_rate=('status', lambda s: (s=='maintenance').mean()),
    avg_vib=('vibration_mm_s','mean'),
    max_vib=('vibration_mm_s','max'),
    avg_temp=('temperature_c','mean'),
    max_temp=('temperature_c','max'),
    low_units_rate=('units_produced', lambda x: (x==0).mean())
).reset_index()
# Create composite attention: higher fault+maintenance, higher max vibration/temp, higher zero production
summary['attention_score'] = (
    5*summary['fault_rate'] + 2*summary['maintenance_rate'] +
    0.8*(summary['max_vib']/summary['avg_vib']) +
    0.2*(summary['max_temp']/summary['avg_temp']) +
    3*summary['low_units_rate']
)
summary.sort_values('attention_score', ascending=False, inplace=True)
print(summary[['machine_id','n_obs','fault_events','maintenance_events','fault_rate','maintenance_rate','avg_vib','max_vib','avg_temp','max_temp','low_units_rate','attention_score']].head(5).to_string(index=False))

# identify main drivers: for top machine, compare metrics during fault/maintenance vs running
