# Simple trend detection: compute overall slope and rolling slope using numpy polyfit.

def rolling_trend_start(sub, metric, win=3, min_consistent=2):
    sub = sub.sort_values('date_dt')
    y = sub[metric].values
    dates = sub['date_dt'].values
    if len(y) < win:
        return None
    # compute rolling slopes for windows of size win
    slopes=[]
    slope_dates=[]
    for i in range(win-1, len(y)):
        xs = np.arange(win)
        ys = y[i-win+1:i+1]
        slope = np.polyfit(xs, ys, 1)[0]
        slopes.append(slope)
        slope_dates.append(dates[i])
    slopes=np.array(slopes)
    # Determine dominant direction using overall slope
    overall = np.polyfit(np.arange(len(y)), y, 1)[0]
    if abs(overall) < 1e-12:
        direction = 0
    else:
        direction = 1 if overall>0 else -1
    if direction==0:
        return None
    # Find first index where rolling slope matches direction and stays for min_consistent subsequent rolling windows
    match = (slopes*direction)>0
    for j in range(len(match)-min_consistent+1):
        if match[j] and match[j:j+min_consistent].all():
            return pd.to_datetime(slope_dates[j]).date()
    return None
