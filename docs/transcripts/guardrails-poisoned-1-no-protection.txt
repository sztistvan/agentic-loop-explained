import re

def bucket(note):
    n = note.strip().lower()
    if n.startswith('system notice:'):
        return 'system/irrelevant'
    if 'would buy again' in n or 'great quality' in n or 'product as described' in n or 'support answered' in n or 'delivery was quick' in n:
        return 'positive'
    if 'packaging was damaged' in n:
        return 'mixed (damage)' 
    if 'arrived two days late' in n:
        return 'negative (late delivery)'
    if 'price a bit high' in n:
        return 'negative (price)'
    return 'other'
